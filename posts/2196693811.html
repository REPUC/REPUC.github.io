# Nginx 综合实战：LNMP 项目

本文是系列的收官篇，将前文所有知识点综合运用到真实生产项目：基于 LNMP 架构搭建 WordPress 博客、可道云私有云、ShopXO 电商，以及 phpMyAdmin 负载均衡 + Redis 会话保持。这些案例完整演示了 Nginx + PHP-FPM + MySQL + Redis 的协同工作。

# 一、项目实战：利用 LNMP 实现 WordPress 站点搭建

环境规划：

  • Web 服务器：CentOS7，部署 Nginx、php-fpm、运行 Web 服务；
  • 数据库：CentOS7，部署 MySQL、Redis 服务。

# 1.1 部署数据库（MySQL）

基于 Shell 脚本部署 MySQL（去除个人信息）：

# 前置：关闭防火墙
systemctl disable --now firewalld

# 准备 MySQL 源码包（mysql-8.0.19-linux-glibc2.12-x86_64.tar.xz）
vim install_mysql5.7or8.0_for_centos.sh
#!/bin/bash
SRC_DIR=`pwd`
MYSQL='mysql-8.0.19-linux-glibc2.12-x86_64.tar.xz'
MYSQL_ROOT_PASSWORD='<root密码>'

check () {
    [ $UID -ne 0 ] && { echo "当前用户不是root,安装失败"; exit 1; }
    cd $SRC_DIR
    [ ! -e $MYSQL ] && { echo "缺少${MYSQL}文件"; exit; }
    [ -e /usr/local/mysql ] && { echo "数据库已存在，安装失败"; exit; }
}

install_mysql(){
    yum -y -q install libaio numactl-libs
    tar xf $MYSQL -C /usr/local/
    MYSQL_DIR=`echo $MYSQL| sed -nr 's/^(.*[0-9]).*/\1/p'`
    ln -s /usr/local/$MYSQL_DIR /usr/local/mysql
    chown -R root.root /usr/local/mysql/
    id mysql &> /dev/null || { useradd -s /sbin/nologin -r mysql; }
    echo 'PATH=/usr/local/mysql/bin/:$PATH' > /etc/profile.d/mysql.sh
    . /etc/profile.d/mysql.sh
    ln -s /usr/local/mysql/bin/* /usr/bin/
    cat > /etc/my.cnf <<-EOF
[mysqld]
server-id=`hostname -I|cut -d. -f4`
log-bin
datadir=/data/mysql
socket=/data/mysql/mysql.sock
log-error=/data/mysql/mysql.log
pid-file=/data/mysql/mysql.pid
[client]
socket=/data/mysql/mysql.sock
EOF
    [ -d /data ] || mkdir /data
    mysqld --initialize-insecure --user=mysql --datadir=/data/mysql
    cp /usr/local/mysql/support-files/mysql.server /etc/init.d/mysqld
    chkconfig --add mysqld && chkconfig mysqld on
    service mysqld start
    sleep 3
    mysqladmin -uroot password $MYSQL_ROOT_PASSWORD
}
check
install_mysql

# 1.3 部署 Nginx

编译安装 Nginx：

systemctl disable --now firewalld
yum -y install gcc pcre-devel openssl-devel zlib-devel
cd /usr/local/src && wget http://nginx.org/download/nginx-1.18.0.tar.gz
tar xf nginx-1.18.0.tar.gz && cd nginx-1.18.0/

./configure --prefix=/apps/nginx --user=www --group=www \
--with-http_ssl_module --with-http_v2_module --with-http_realip_module \
--with-http_stub_status_module --with-http_gzip_static_module \
--with-pcre --with-stream --with-stream_ssl_module --with-stream_realip_module
make -j 4 && make install

准备 service 文件并启动：

[Unit]
Description=nginx - high performance web server
Documentation=http://nginx.org/en/docs/
After=network-online.target remote-fs.target nss-lookup.target
Wants=network-online.target

[Service]
Type=forking
PIDFile=/apps/nginx/run/nginx.pid
ExecStart=/apps/nginx/sbin/nginx -c /apps/nginx/conf/nginx.conf
ExecReload=/bin/kill -s HUP $MAINPID
ExecStop=/bin/kill -s TERM $MAINPID

[Install]
WantedBy=multi-user.target
mkdir /apps/nginx/run
vim /apps/nginx/conf/nginx.conf   # pid /apps/nginx/run/nginx.pid;
systemctl daemon-reload
systemctl enable --now nginx

配置 Nginx 支持 FastCGI：

worker_processes  auto;
pid       /apps/nginx/run/nginx.pid;
events {
    worker_connections  1024;
}
http {
    include       mime.types;
    default_type application/octet-stream;
    sendfile       on;
    keepalive_timeout  65;
    server {
        listen       80;
        server_name blog.example.org;              # 站点域名
        location / {
            root   /data/nginx/wordpress;          # WordPress 源码目录
            index index.php index.html index.htm;  # 默认主页含 index.php
        }
        error_page   500 502 503 504 /50x.html;
        location = /50x.html { root html; }
        location ~ \.php$ {                         # PHP 请求转发 PHP-FPM
            root           /data/nginx/wordpress;
            fastcgi_pass   127.0.0.1:9000;
            fastcgi_index index.php;
            fastcgi_param SCRIPT_FILENAME  $document_root$fastcgi_script_name;
            include       fastcgi_params;
        }
        location ~ ^/(ping|pm_status)$ {            # 状态页
            include fastcgi_params;
            fastcgi_pass 127.0.0.1:9000;
            fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        }
    }
}
ln -s /apps/nginx/sbin/nginx /usr/bin/nginx
nginx -t && systemctl reload nginx

mkdir -p /data/nginx/wordpress
vim /data/nginx/wordpress/test.php

# 访问 http://blog.example.org/ping、/pm_status、/test.php 验证

# 1.5 PHP 扩展 Session 模块支持 Redis

将 PHP Session 存放到 Redis，实现负载均衡下的会话共享。

编译安装 phpredis 模块：

cd /usr/local/src
wget http://pecl.php.net/get/redis-5.3.1.tgz
tar xf redis-5.3.1.tgz && cd redis-5.3.1/
/apps/php74/bin/phpize    # 生成 configure（需先装 autoconf）
./configure --with-php-config=/apps/php74/bin/php-config
make -j 8 && make install
# 模块文件生成于 /apps/php74/lib/php/extensions/.../redis.so

编辑 PHP 配置文件支持 Redis：

vim /etc/php.ini
extension=redis.so     # 启用 redis 扩展
systemctl restart php-fpm

安装和配置 Redis 服务：

yum install epel-release -y && yum -y install redis
vim /etc/redis.conf
bind 0.0.0.0
protected-mode no
requirepass '<密码>'
systemctl enable --now redis

配置 PHP 使用 Redis 保存 Session：

vim /etc/php.ini
[Session]
session.save_handler = redis
session.save_path = "tcp://192.168.80.17:6379?auth=<密码>"
systemctl restart php-fpm

测试 Session 存入 Redis：

// /data/nginx/wordpress/session.php

# 四、项目实战：包安装利用 LNMP 实现 phpMyAdmin 的负载均衡并利用 Redis 实现会话保持

核心思想：负载均衡把访问量分担到多台后端服务器；但登录后每次请求可能分配到不同后端，各服务器有自己的 Session，导致每次刷新都要重新登录。Session 共享解决此问题——浏览器端存 Cookie，每次请求自动带 Cookie，服务端以 Cookie 为 key 查找 Session。同一域名下用户 Cookie 相同，无论请求分配到哪台服务器，Cookie 对应 Session 唯一，因此只要多台业务服务器访问同一个 Redis 存放 Session 即可。

systemctl restart mysql.service
mysql> create user admin@'10.0.0.%' identified with mysql_native_password by '<密码>';
mysql> grant all on *.* to admin@'10.0.0.%';

# 4.6 配置 Nginx 反向代理实现负载均衡

在负载均衡服务器上配置：

groupadd -g 666 -r www
useradd -u 666 -g www -s /sbin/nologin -r www
bash install_nginx.sh
# conf.d/proxy-www.example.org.conf
upstream webservers {
    server 192.168.80.18:80;    # 后端服务器
    server 192.168.80.28:80;
}

server {
    listen 80;
    server_name www.example.org;
    return 302 https://$server_name$request_uri;   # HTTP 跳转 HTTPS
}

server {
    listen 443 ssl http2;
    server_name www.example.org;
    ssl_certificate    /apps/nginx/ssl/www.example.org.pem;
    ssl_certificate_key /apps/nginx/ssl/www.example.org.key;
    ssl_session_cache shared:sslcache:20m;
    ssl_session_timeout 10m;
    location / {
        proxy_pass http://webservers;          # 负载均衡
        proxy_set_header Host $http_host;      # 转发请求头
    }
}
ln -s /apps/nginx/sbin/nginx /usr/bin/nginx
nginx -t && systemctl restart nginx && systemctl enable --now nginx
