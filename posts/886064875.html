# Nginx 四层代理与 FastCGI

本文讲解 Nginx 的四层（TCP/UDP）负载均衡与动态资源处理能力：stream 模块实现 TCP/UDP 四层代理、FastCGI 协议与 PHP-FPM 的集成、动静分离。这是 Nginx 处理非 HTTP 协议与动态请求的关键。

# 一、实现 Nginx 四层负载均衡

Nginx 在 1.9.0 版本开始支持 TCP 模式负载均衡，1.9.13 版本开始支持 UDP 协议负载（UDP 主要用于 DNS 域名解析）。其基于 ngx_stream_proxy_module 模块实现 TCP 负载，基于 ngx_stream_upstream_module 实现后端服务器分组转发、权重分配、状态监测、调度算法。

编译安装需指定 --with-stream 选项才能支持 stream 模块。

# 1.1 TCP 协议负载均衡配置参数

配置编写在 stream 语句块（Context: main）。

stream {
    upstream backend {
        hash $remote_addr consistent;   # 调度算法，后端可用时生效
        server backend1.example.com:12345 weight=5;
        server 127.0.0.1:12345 max_fails=3 fail_timeout=30s;
        server unix:/tmp/backend3;
    }

    upstream dns {
        server 10.0.0.1:53535;
        server dns.example.com:53;
    }

    server {
        listen 12345;                 # 监听 IP:PORT
        proxy_connect_timeout 1s;     # 连接超时
        proxy_timeout 3s;             # 转发超时
        proxy_pass backend;           # 转发到服务器组
    }

    server {
        listen 127.0.0.1:53 udp reuseport;   # UDP 监听
        proxy_timeout 20s;
        proxy_pass dns;
    }

    server {
        listen [::1]:12345;
        proxy_pass unix:/tmp/stream.socket;
    }

    server {
        listen 2222;
        proxy 10.0.0.8:22;            # 只实现四层代理，无负载均衡
    }
}

要点：

  • stream 块与 http 块平级，配置在主配置文件顶层；
  • listen 指定监听 IP 和端口；
  • proxy_pass 转发到 upstream 组（负载均衡）或直接地址（纯代理）；
  • proxy_connect_timeout 连接超时、proxy_timeout 转发超时。

# 1.2 TCP 负载均衡实例：Redis

后端安装 Redis：

# 安装两台 redis 服务器
yum install redis -y
sed -i '/^bind /c bind 0.0.0.0' /etc/redis.conf
systemctl enable --now redis
ss -tnl | grep 6379

Nginx 配置：

# 主配置文件增加 include（与 http 模块平级，可写在最尾行）
vim /apps/nginx/conf/nginx.conf
include /apps/nginx/conf/tcp/*.conf;
# conf/tcp/redis_server.conf
stream {
    upstream redis_server {
        # hash $remote_addr consistent;
        server 192.168.80.18:6379 max_fails=3 fail_timeout=30s;
        server 192.168.80.28:6379 max_fails=3 fail_timeout=30s;
    }
    server {
        listen 192.168.80.8:6379;       # 监听本机端口
        proxy_connect_timeout 3s;
        proxy_timeout 3s;
        proxy_pass redis_server;        # 负载均衡调度
    }
}
nginx -t && nginx -s reload
ss -tnl | grep 6379   # 验证 192.168.80.8:6379 监听

# 测试通过 Nginx 负载连接 redis
redis-cli -h 192.168.80.8 set name example
OK
redis-cli -h 192.168.80.8 get name
"example"

# 1.3 TCP 负载均衡实例：MySQL

后端安装 MySQL：

# 修改后端主机名
hostnamectl set-hostname mysql-server1.example.org
hostnamectl set-hostname mysql-server2.example.org

# 安装 MySQL
yum -y install mariadb-server
systemctl enable --now mariadb

# 创建可远程访问的用户（注意修改 IP 段）
MariaDB [(none)]> create user example@'192.168.80.%' identified by '<密码>';
MariaDB [(none)]> FLUSH PRIVILEGES;

Nginx 配置：

# conf/tcp/mysql_server.conf
stream {
    upstream mysql_server {
        least_conn;
        server 192.168.80.18:3306 max_fails=3 fail_timeout=30s;
        server 192.168.80.28:3306 max_fails=3 fail_timeout=30s;
    }
    server {
        listen 192.168.80.8:3306;       # Nginx 伪监听 3306
        proxy_connect_timeout 6s;
        proxy_timeout 15s;
        proxy_pass mysql_server;        # 负载均衡调度
    }
}
nginx -t && nginx -s reload
ss -tnl | grep 3306   # Nginx 反向代理服务器伪监听 3306

# 测试通过 nginx 负载连接后端 MySQL
mysql -uexample -p'<密码>' -h192.168.80.8 -e 'show variables like "hostname"'
# 停止一台 MySQL 后，再次访问只会由存活的 MySQL 响应（状态监测生效）

# 1.4 UDP 协议负载均衡实例：DNS

stream {
    upstream dns_servers {
        server 10.0.0.18:53;
        server 10.0.0.28:53;
    }
    server {
        listen 53 udp;              # 监听本机端口
        proxy_pass dns_servers;     # 负载均衡调度
    }
}

# 二、实现 FastCGI

# 2.1 LNMP 优势与 CGI 由来

LNMP 优势：Nginx 将客户端发送的 PHP 相关请求转发至后端 PHP 服务器（PHP-FPM）处理；Nginx 基于 FastCGI 协议与 PHP-FPM 通信，将请求数据（GET/POST 参数）传递给后端。

CGI 由来：最早的 Web 服务器只能响应浏览器请求并返回静态 HTML。后期出现动态技术（PHP/Java/Python），但 Nginx/Apache 不能直接运行这些文件。Apache 通过安装 PHP 模块扩展动态处理能力；Nginx 通过与第三方基于协议实现——将客户端请求转发给第三方服务处理，第三方新建进程处理请求，完成后返回数据给 Nginx 并回收进程，Nginx 再返回客户端。这个约定就是通用网关接口（CGI，Common Gateway Interface），是 Web 服务器和外部应用程序之间传递信息的标准化接口。

为什么会有 FastCGI：CGI 效率低，Web 服务器每收到一个请求都创建 CGI 进程，PHP 解析器都解析 php.ini、初始化环境，请求结束关闭进程，每个进程都重复这些操作。FastCGI 每次处理完请求后不关闭进程，保留进程处理多个请求，每个请求无需重新创建进程，大幅提升处理效率。

什么是 PHP-FPM：PHP-FPM（FastCGI Process Manager，FastCGI 进程管理器）是实现 FastCGI 的程序，提供进程管理。进程包括 Master 进程（只有一个，监听端口，接受 Web 服务器请求）和 Worker 进程（多个，每个嵌入一个 PHP 解析器处理 PHP 代码）。

# 2.2 FastCGI 配置指令

Nginx 基于 ngx_http_fastcgi_module 通过 FastCGI 协议将客户端请求转发至 php-fpm 处理。

fastcgi_pass address;
# 转发请求到后端 fastcgi server，address 为后端地址，可用 location、if in location
# 示例
fastcgi_pass localhost:9000;
fastcgi_pass unix:/tmp/fastcgi.socket;    # 注意文件权限

fastcgi_index name;
# fastcgi 默认主页资源，如 fastcgi_index index.php;

fastcgi_param parameter value [if_not_empty];
# 设置传递给 FastCGI 服务器的参数值，可为文本、变量或组合
fastcgi_param REMOTE_ADDR $remote_addr;     # 客户端源 IP
fastcgi_param REMOTE_PORT $remote_port;     # 客户端源端口
fastcgi_param SERVER_ADDR $server_addr;     # 服务器 IP
fastcgi_param SERVER_PORT $server_port;     # 服务器端口
fastcgi_param SERVER_NAME $server_name;     # server name

Nginx 默认配置示例：

location ~ \.php$ {
    root           /scripts;
    fastcgi_pass   127.0.0.1:9000;
    fastcgi_index index.php;
    fastcgi_param SCRIPT_FILENAME  $document_root$fastcgi_script_name;   # 默认脚本路径
    # fastcgi_param SCRIPT_FILENAME /scripts$fastcgi_script_name;       # 此写法无需上面 root
    fastcgi_param HTTPS on;           # 前端代理用 https 时，后端 http 服务器需加此项，否则页面异常
    include       fastcgi_params;     # 系统默认提供
}

FastCGI 缓存指令（生产慎用，可能导致源码更新失败）：

fastcgi_cache_path path [levels=levels] [use_temp_path=on|off]
keys_zone=name:size [inactive=time] [max_size=size] ...;
# path 缓存位置；max_size 磁盘缓存上限；levels 缓存目录层级；keys_zone 缓存名称及内存大小；inactive 缓存有效时间（默认 10 分钟）

fastcgi_cache zone | off;          # 调用缓存，Context: http, server, location
fastcgi_cache_key string;          # 缓存键，如 fastcgi_cache_key $request_uri;
fastcgi_cache_methods GET|HEAD|POST ...;   # 对哪些请求方法缓存
fastcgi_cache_min_uses number;     # 缓存项在 inactive 内至少被访问次数才视为活动项
fastcgi_keep_conn on|off;          # 收到响应后是否关闭连接，建议启用长连接
fastcgi_cache_valid [code ...] time;    # 不同响应码缓存时长
fastcgi_hide_header field;         # 隐藏响应头
fastcgi_pass_header field;         # 透传响应头（默认不传 Status、X-Accel-...）

# 2.3 实战：Nginx 与 PHP-FPM 在同一服务器

# PHP 环境准备

Ubuntu：

apt -y install php-fpm
php-fpm8.1 -v
systemctl status php8.1-fpm
ll /run/php/php-fpm.sock    # socket 文件

若 nginx 用 socket 与 PHP 通信，需让 nginx 和 PHP-FPM 使用相同用户（如 www-data），否则 502；若通过 9000 端口通信则无需。

CentOS（Base 源）：

yum install php-fpm php-mysqlnd php-json -y
systemctl enable --now php-fpm
ps -ef | grep php-fpm   # master + 多个 pool www worker

# PHP 相关配置优化

Ubuntu：

vim /etc/php/8.1/fpm/php.ini
date.timezone = Asia/Shanghai      # 时区
post_max_size = 100M               # POST 最大数据
upload_max_filesize = 100M         # 上传文件最大

vim /etc/php/8.1/fpm/pool.d/www.conf
listen = 127.0.0.1:9000            # 监听 9000 端口
;listen = /run/php/php8.1-fpm.sock  # 注释，不用 socket
pm.status_path = /pm_status        # 状态页
ping.path = /ping                   # 检查 PHP-FPM 是否正常
systemctl restart php8.1-fpm.service

CentOS：

vim /etc/php.ini
date.timezone = Asia/Shanghai
post_max_size = 100M
upload_max_filesize = 100M

vim /etc/php-fpm.d/www.conf
[www]
user = nginx          # 修改
group = nginx         # 修改
listen = 127.0.0.1:9000
listen.allowed_clients = 127.0.0.1

pm = dynamic
pm.max_children = 50
pm.start_servers = 5
pm.min_spare_servers = 5
pm.max_spare_servers = 35
pm.status_path = /pm_status
ping.path = /ping
ping.response = ping-pong
slowlog = /var/log/php-fpm/www-slow.log
php_admin_value[error_log] = /var/log/php-fpm/www-error.log
php_admin_flag[log_errors] = on
php_value[session.save_handler] = files
php_value[session.save_path] = /var/lib/php/session

# 如存在 Redis，修改为：
php_value[session.save_handler] = redis
php_value[session.save_path] = "tcp://redis-server:6379"

systemctl restart php-fpm

# 准备 PHP 测试页面

mkdir -p /data/php
vim /data/php/index.php
<?php phpinfo(); ?>

# 测试连接数据库
vim /data/php/mysql.php
<?php
    $servername = "localhost";
    $username = "root";
    $password = "<密码>";
    $conn = mysqli_connect($servername, $username, $password);
    if (!$conn) { die("php连接MySQL数据库失败: " . mysqli_connect_error()); }
    echo "php连接MySQL数据库成功!";
?>

yum install php -y
php /data/php/index.php
php /data/php/mysql.php

# Nginx 配置转发

server {
    listen 80;
    server_name www.example.org;
    # root /data/php;    # 需与 location 的 root 一致
    index index.php index.html;

    location ~ \.php$|pm_status|ping {
        root /data/php;                            # 存放 PHP 文件目录，$document_root 调用此行
        # fastcgi_pass unix:/run/php-fpm/www.sock;  # socket 方式
        fastcgi_pass 127.0.0.1:9000;               # 端口方式
        fastcgi_index index.php;
        fastcgi_param SCRIPT_FILENAME  $document_root$fastcgi_script_name;
        include       fastcgi_params;
        # fastcgi_param HTTPS $https if_not_empty;  # 支持 https 的应用需此项
    }
}
nginx -t && nginx -s reload

# 访问验证

访问 http://www.example.org/index.php 显示 phpinfo 页面，表示 Nginx 已成功连接 PHP-FPM。

常见错误：

  • File not found：访问路径不对；
  • 502 Bad Gateway：php-fpm 处理超时、服务停止、无法连接或请求超时。

# PHP-FPM 运行状态页面

location ~ ^/(ping|pm_status)$ {
    fastcgi_pass unix:/run/php-fpm/www.sock;
    # fastcgi_pass 127.0.0.1:9000;
    fastcgi_param PATH_TRANSLATED $document_root$fastcgi_script_name;
    include fastcgi_params;
}
curl http://www.example.org/ping        # ping-pong
curl http://www.example.org/pm_status   # 返回 pool、process manager、start time、accepted conn、idle/active/total processes 等
curl http://www.example.org/pm_status?full
curl http://www.example.org/pm_status?html
curl http://www.example.org/pm_status?json

# 2.4 实战：Nginx 与 PHP 不在同一服务器

Nginx 处理静态请求，转发动态请求到后端 PHP-FPM 服务器，PHP 代码放在后端。

后端安装 PHP-FPM（较新版本）：

dnf -y install php-fpm php-mysqlnd php-json
rpm -ql php-fpm    # 验证安装路径

修改 PHP-FPM 监听（默认仅 127.0.0.1:9000，无法远程）：

vim /etc/php-fpm.d/www.conf
;listen = /run/php-fpm/www.sock        # 注释
listen = 9000                          # 指定监听端口
;listen.allowed_clients = 127.0.0.1    # 注释，允许远程

Nginx 配置转发：

location ~ \.php$ {
    root           /data/php;
    fastcgi_pass   10.0.0.18:9000;          # 后端 PHP-FPM
    fastcgi_index index.php;
    fastcgi_param SCRIPT_FILENAME  $document_root$fastcgi_script_name;
    include       fastcgi_params;
}
location ~ ^/(ping|pm_status)$ {
    fastcgi_pass 10.0.0.18:9000;
    fastcgi_param PATH_TRANSLATED $document_root$fastcgi_script_name;
    include fastcgi_params;
}

# 2.5 实现动静分离

要求：将客户端对除 PHP 以外资源的访问转发至后端服务器 10.0.0.28。

location / {
    proxy_pass http://10.0.0.28;       # 静态资源转发至后端
    index index.html;
}
location ~ \.php$ {
    root /data/php;
    fastcgi_pass   10.0.0.18:9000;      # 动态 PHP 转发至 PHP-FPM
    fastcgi_index index.php;
    fastcgi_param SCRIPT_FILENAME  $document_root$fastcgi_script_name;
    include       fastcgi_params;
}
# 后端 10.0.0.28 安装 httpd 提供静态资源
dnf -y install httpd
systemctl enable --now httpd
mkdir /var/www/html/images
wget -O /var/www/html/images/logo.jpg <图片URL>
